📁 Manual JSON Export

Export your data as a JSON file by running this code in the browser console:

// Export all recorded data as JSON
function exportData() {
  const data = {
    timestamp: new Date().toISOString(),
    metrics: {
      heartRate: document.getElementById('hr-value').textContent,
      hrv: document.getElementById('hrv-value').textContent,
      respiratoryRate: document.getElementById('rr-value').textContent,
      stress: document.getElementById('stress-value').textContent
    },
    demographics: {
      age: document.getElementById('age-value').textContent,
      gender: document.getElementById('gender-value').textContent,
      expression: document.getElementById('expression-value').textContent,
      ethnicity: document.getElementById('ethnicity-value').textContent
    },
    quality: {
      snr: document.getElementById('snr-value').textContent,
      motion: document.getElementById('motion-score').textContent,
      lighting: document.getElementById('lighting-score').textContent
    },
    rawData: window.analyzer ? analyzer.recordingData : []
  };
  
  const blob = new Blob([JSON.stringify(data, null, 2)], {type: 'application/json'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `fivelidz-export-${Date.now()}.json`;
  a.click();
}

exportData();

📈 CSV Export for Analysis

Convert your data to CSV format for use in Excel, Google Sheets, or data analysis tools:

// Export as CSV
function exportCSV() {
  if (!window.analyzer || !analyzer.recordingData.length) {
    alert('No data to export. Start recording first.');
    return;
  }
  
  let csv = 'Timestamp,PPG Signal,Red Channel,Green Channel,Blue Channel\\n';
  
  analyzer.recordingData.forEach(row => {
    csv += `${row.timestamp},${row.ppg},${row.channels.red},${row.channels.green},${row.channels.blue}\\n`;
  });
  
  const blob = new Blob([csv], {type: 'text/csv'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `fivelidz-data-${Date.now()}.csv`;
  a.click();
}

exportCSV();

storage Data Structure

Recording Data Object

  • timestamp: ISO 8601 datetime when measurement was taken
  • duration: Total recording duration in seconds
  • samples: Number of data points collected
  • metrics: Object containing:
    • heartRate: Beats per minute (BPM)
    • hrv: Heart rate variability (RMSSD in ms)
    • respiratoryRate: Breaths per minute
    • stress: Stress level (Low/Normal/Moderate/High)
  • demographics: Object containing:
    • age: Estimated age ± 3 years
    • gender: Male/Female classification
    • expression: Dominant facial expression
    • ethnicity: Broad ethnic category
  • rawData: Array of raw PPG measurements:
    • timestamp: Unix timestamp (ms)
    • ppg: Weighted PPG signal value
    • channels: RGB channel averages

code API Integration

Send your data to your own server or API endpoint:

// Send data to your API
async function sendToAPI() {
  const data = {
    timestamp: new Date().toISOString(),
    metrics: {
      heartRate: document.getElementById('hr-value').textContent,
      hrv: document.getElementById('hrv-value').textContent,
      respiratoryRate: document.getElementById('rr-value').textContent,
      stress: document.getElementById('stress-value').textContent
    }
  };
  
  try {
    const response = await fetch('https://your-api.com/endpoint', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer YOUR_API_KEY'
      },
      body: JSON.stringify(data)
    });
    
    if (response.ok) {
      console.log('Data sent successfully');
    } else {
      console.error('Failed to send data');
    }
  } catch (error) {
    console.error('Error:', error);
  }
}

query_stats Data Analysis Examples 

// Calculate average heart rate from session
function getAverageHR() {
  if (!window.analyzer || !analyzer.hrTrend.length) return 0;
  
  const sum = analyzer.hrTrend.reduce((a, b) => a + b, 0);
  return Math.round(sum / analyzer.hrTrend.length);
}

// Find heart rate variability range
function getHRVRange() {
  // Get min and max HRV values from session
  const hrvValues = analyzer.recordingData
    .map(d => d.hrv)
    .filter(v => v > 0);
  
  return {
    min: Math.min(...hrvValues),
    max: Math.max(...hrvValues),
    avg: Math.round(hrvValues.reduce((a, b) => a + b) / hrvValues.length)
  };
}

// Calculate session statistics
function getSessionStats() {
  return {
    duration: analyzer.recordingData.length / 30, // seconds at 30fps
    samples: analyzer.recordingData.length,
    avgHeartRate: getAverageHR(),
    hrvRange: getHRVRange(),
    signalQuality: document.getElementById('snr-value').textContent
  };
}

console.log(getSessionStats());

privacy_tip Privacy & Data Security

  • All processing happens locally in your browser
  • No data is sent to external servers unless you explicitly export it
  • Recording data is stored only in browser memory (RAM)
  • Data is cleared when you refresh or close the page
  • Demographics data should be handled with care and respect for privacy
  • Consider anonymizing data before sharing or storing
  • Follow GDPR/CCPA guidelines if collecting data from others

Python Analysis

import json
import pandas as pd
import matplotlib.pyplot as plt

# Load exported JSON file
with open('fivelidz-export.json', 'r') as f:
    data = json.load(f)

# Convert to DataFrame
df = pd.DataFrame(data['rawData'])
df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')

# Plot PPG signal
plt.figure(figsize=(12, 6))
plt.plot(df['timestamp'], df['ppg'])
plt.xlabel('Time')
plt.ylabel('PPG Signal')
plt.title('Fivelidz PPG Recording')
plt.show()

# Calculate statistics
print(f"Average PPG: {df['ppg'].mean():.2f}")
print(f"Signal variance: {df['ppg'].var():.2f}")
print(f"Recording duration: {(df['timestamp'].max() - df['timestamp'].min()).total_seconds():.1f}s")

help Troubleshooting

Common Issues

  • No data to export: Ensure you've started recording and collected at least 60 samples
  • Console commands not working: Check that 'analyzer' object exists (refresh page if needed)
  • CSV has missing values: Some metrics require minimum recording duration (15+ seconds)
  • Browser blocks download: Check your browser's download settings and popup blocker
  • Data looks incorrect: Verify signal quality indicators are good (SNR >10dB)
